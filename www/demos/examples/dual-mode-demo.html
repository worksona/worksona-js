Mode Comparison

Library Mode (Direct)

  • Import worksona.min.js
  • Direct JavaScript API
  • Browser or Node.js
  • Full type definitions
  • Event system access
  • No network overhead
const worksona = new Worksona({ apiKeys: { openai: 'sk-...' } }); await worksona.loadAgent({ id: 'assistant', config: { model: 'gpt-5' } }); const answer = await worksona.chat( 'assistant', 'Your question' );

API Mode (REST)

  • HTTP REST endpoints
  • Works from any language
  • URL query parameters
  • File upload support
  • Webhook integrations
  • Easy external integration
// Simple GET request const response = await fetch( 'http://localhost:3000/api/query' + '?q=Your+question&model=gpt-5' ); const data = await response.json(); console.log(data.answer); // Or use cURL curl "http://localhost:3000/api/query?q=Hello"

Code Examples

JavaScript - Library Mode

// Initialize Worksona const worksona = new Worksona({ apiKeys: { openai: 'your-api-key', anthropic: 'your-api-key' } }); // Load an agent with GPT-5 await worksona.loadAgent({ id: 'gpt5-agent', name: 'GPT-5 Assistant', config: { provider: 'openai', model: 'gpt-5', temperature: 0.7 } }); // Chat with agent const response = await worksona.chat('gpt5-agent', 'Explain quantum computing'); console.log(response); // Process image const analysis = await worksona.processImage('gpt5-agent', imageUrl, { prompt: 'What is in this image?' });

JavaScript - API Mode

// Simple query const response = await fetch( 'http://localhost:3000/api/query?q=What+is+AI&model=gpt-5' ); const data = await response.json(); console.log(data.answer); // Agent-specific query const response2 = await fetch( 'http://localhost:3000/api/agents/my-agent/query?q=Hello' ); // Batch queries const batch = await fetch('http://localhost:3000/api/query/batch', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ queries: [ { query: 'Translate: Hello', model: 'gpt-4o' }, { query: 'Summarize: Long text...', model: 'gpt-5' } ] }) });

Python - API Mode

import requests # Simple query response = requests.get( 'http://localhost:3000/api/query', params={'q': 'What is AI', 'model': 'gpt-5'} ) data = response.json() print(data['answer']) # Agent-specific query response = requests.get( 'http://localhost:3000/api/agents/analyst/query', params={'q': 'Analyze this data'} ) # Upload file for analysis with open('document.pdf', 'rb') as f: files = {'file': f} response = requests.post( 'http://localhost:3000/api/documents/analyze', files=files, data={'task': 'summarize'} ) # Batch processing response = requests.post( 'http://localhost:3000/api/query/batch', json={ 'queries': [ {'query': 'Question 1', 'model': 'gpt-5'}, {'query': 'Question 2', 'model': 'gpt-4o'} ] } )

cURL - API Mode

# Simple query curl "http://localhost:3000/api/query?q=What+is+AI&model=gpt-5" # Agent-specific query curl "http://localhost:3000/api/agents/legal-expert/query?q=Review+contract" # Upload and analyze image curl -X POST http://localhost:3000/api/images/analyze \ -F "image=@photo.jpg" \ -F "prompt=What is in this image?" # OCR document curl -X POST http://localhost:3000/api/documents/ocr \ -F "file=@invoice.jpg" # Extract invoice data curl -X POST http://localhost:3000/api/agents/invoice-agent/process \ -F "file=@invoice.pdf" \ -F "schema=invoice" # Batch queries curl -X POST http://localhost:3000/api/query/batch \ -H "Content-Type: application/json" \ -d '{ "queries": [ {"query": "Translate: Hello", "model": "gpt-4o"}, {"query": "Summarize: Text...", "model": "gpt-5"} ] }' # Webhook endpoint curl -X POST http://localhost:3000/api/webhook/support-bot \ -H "Content-Type: application/json" \ -d '{"text": "How do I reset my password?"}'

Node.js - Both Modes

// Library Mode (Direct) const Worksona = require('./worksona.js'); const worksona = new Worksona({ apiKeys: { openai: process.env.OPENAI_API_KEY, anthropic: process.env.ANTHROPIC_API_KEY } }); // Load agent await worksona.loadAgent({ id: 'assistant', config: { provider: 'openai', model: 'gpt-5' } }); // Query const answer = await worksona.chat('assistant', 'Your question'); // --- // API Mode (HTTP requests) const fetch = require('node-fetch'); // Simple query const response = await fetch( 'http://localhost:3000/api/query?q=Hello&model=gpt-5' ); const data = await response.json(); // Upload file const FormData = require('form-data'); const fs = require('fs'); const formData = new FormData(); formData.append('file', fs.createReadStream('document.pdf')); const fileResponse = await fetch( 'http://localhost:3000/api/documents/analyze', { method: 'POST', body: formData } );
